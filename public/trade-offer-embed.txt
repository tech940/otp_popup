<script>
(function () {
  var POPUP_ORIGIN = 'https://www.amfordofjefferson.com';
  try {
    if (!document.querySelector('link[rel="preconnect"][href="' + POPUP_ORIGIN + '"]')) {
      var _tp1 = document.createElement('link'); _tp1.rel = 'preconnect'; _tp1.href = POPUP_ORIGIN; document.head.appendChild(_tp1);
      var _tp2 = document.createElement('link'); _tp2.rel = 'dns-prefetch'; _tp2.href = POPUP_ORIGIN; document.head.appendChild(_tp2);
    }
    if (!document.getElementById('otp-spin-style')) {
      var _ss = document.createElement('style'); _ss.id = 'otp-spin-style';
      _ss.textContent = '@keyframes otpSpin{0%{transform:translate(-50%,-50%) rotate(0deg);}100%{transform:translate(-50%,-50%) rotate(360deg);}}';
      document.head.appendChild(_ss);
    }
  } catch (e) {}
  var ENQUIRY_KEYS = ["offer_dismissed_until","otp_verified_user","otp_enquiry_start"];

  /* ─────────────────────────────────────────────
     ASC EVENTS  (Automotive Standards Council)
     This popup opens on a timer, so it raises asc_form_engagment itself. This script
     only relays the popup's ASC events into the page dataLayer:
     asc_form_engagment / asc_form_submission / asc_form_submission_sales.
     Spec: https://docs.google.com/spreadsheets/d/16i1x19MK3yCRXMivGa2WcyOhxlmGgkb9GbE2vAJuJ74
  ───────────────────────────────────────────── */
  /* Only these three event names are ever pushed — a relayed message can never name
     its own GTM event. */
  var ASC_ALLOWED_EVENTS = ['asc_form_engagment', 'asc_form_submission', 'asc_form_submission_sales'];

  /* Every ASC key these events can carry. The full set is pushed on every event so a key
     never carries over: GTM merges each push into one persistent data model, so a key that
     is simply omitted keeps whatever the previous event set — which would attach the last
     vehicle a visitor viewed to an unrelated later lead.
     Keep in sync with ASC_PARAM_KEYS in src/lib/ascEvents.ts. */
  var ASC_PARAM_KEYS = ['event_owner', 'affiliation', 'department', 'page_type', 'page_location',
    'comm_type', 'comm_status', 'form_name', 'form_type', 'element_type', 'element_subtype',
    'element_text', 'element_position', 'event_action', 'event_action_result', 'submission_id',
    'flow_name', 'flow_outcome', 'item_id', 'item_number', 'item_year', 'item_make', 'item_model',
    'item_variant', 'item_color', 'item_type', 'item_fuel_type', 'item_condition', 'item_price',
    'currency'];

  function ascClean(params) {
    var out = {};
    for (var k in params) {
      if (!Object.prototype.hasOwnProperty.call(params, k)) continue;
      var v = params[k];
      if (v === null || v === undefined) continue;
      if (typeof v === 'number') { out[k] = v; continue; }
      var s = String(v).replace(/^\s+|\s+$/g, '');
      if (s === '') continue;
      out[k] = s;
    }
    return out;
  }

  function ascPush(eventName, params) {
    try {
      window.dataLayer = window.dataLayer || [];
      var clean = ascClean(params || {});
      var payload = { event: eventName };
      for (var i = 0; i < ASC_PARAM_KEYS.length; i++) {
        var key = ASC_PARAM_KEYS[i];
        payload[key] = Object.prototype.hasOwnProperty.call(clean, key) ? clean[key] : undefined;
      }
      window.dataLayer.push(payload);
    } catch (e) {}
  }

  function ascPageTypeValue() {
    var path = (window.location.pathname || '').toLowerCase();
    if (path.indexOf('/inventory/') !== -1) return 'item';
    if (path.indexOf('/used-vehicles') !== -1 || path.indexOf('/new-vehicles') !== -1) return 'itemlist';
    if (path === '/' || path === '' || path.slice(-5) === '/home' || path.slice(-6) === '/home/') return 'home';
    return 'unknown';
  }

  /* Relay ASC events posted by the popup iframe into this page's dataLayer. Only known
     event names and known parameter keys are copied across, so a message can neither name
     its own GTM event nor introduce a key of its own. */
  function ascRelay(d) {
    if (!d || typeof d !== 'object' || d.type !== 'ASC_EVENT') return false;
    if (ASC_ALLOWED_EVENTS.indexOf(d.event) === -1) return false;
    var src = (d.params && typeof d.params === 'object') ? d.params : {};
    var params = {};
    for (var i = 0; i < ASC_PARAM_KEYS.length; i++) {
      var key = ASC_PARAM_KEYS[i];
      if (Object.prototype.hasOwnProperty.call(src, key)) params[key] = src[key];
    }
    /* The parent page is authoritative for page context. */
    params.page_location = window.location.href;
    params.page_type = ascPageTypeValue();
    ascPush(d.event, params);
    return true;
  }

  /* Exact-origin match only. A substring test such as indexOf('vercel.app') would trust
     any host containing that text — including free *.vercel.app sites and lookalikes such
     as vercel.app.example.com. localhost is honoured only when this page is itself served
     from localhost, i.e. during local development. */
  function ascTrustedOrigin(origin) {
    var o = String(origin || '').replace(/\/$/, '');
    if (!o) return false;
    if (o === String(POPUP_ORIGIN || '').replace(/\/$/, '')) return true;
    if (o === window.location.origin) return true;
    var LOCAL = /^https?:\/\/(localhost|127\.0\.0\.1)(:\d+)?$/;
    return LOCAL.test(o) && LOCAL.test(window.location.origin);
  }

  /* Bound once per page — every embed on the site shares this guard, so a single submit
     never lands in the dataLayer more than once. */
  if (!window.__ascEventRelayBound) {
    window.__ascEventRelayBound = true;
    window.addEventListener('message', function (ev) {
      if (!ascTrustedOrigin(ev.origin)) return;
      ascRelay(ev.data);
    });
  }

  function clearOldEnquiry() {
    ENQUIRY_KEYS.forEach(function(k){
      try { localStorage.removeItem(k); } catch(_){}
      try { sessionStorage.removeItem(k); } catch(_){}
    });
  }

  function clearEnquiryCookies() {
    var rx = /^(otp_|offer_)/;
    document.cookie.split(";").forEach(function(c) {
      var name = c.split("=")[0].trim();
      if (rx.test(name)) {
        document.cookie = name + "=;expires=Thu, 01 Jan 1970 00:00:00 GMT;path=/";
      }
    });
  }

  function getPageType() {
    var path = (window.location.pathname || '').toLowerCase();
    var isHomePath = path === '/' || path === '' || path === '/index.php' || path === '/index.html';
    var endsWithHomeSlash = path.slice(-6) === '/home/';
    var endsWithHome = path.slice(-5) === '/home';
    if (isHomePath || endsWithHomeSlash || endsWithHome) return 'Home';
    if (path.indexOf('/used-vehicles') !== -1 || path.indexOf('/new-vehicles') !== -1) return 'SRP';
    if (path.indexOf('/inventory/') !== -1) return 'VDP';
    return 'Other';
  }

  function isUnlockPopupOpen() {
    if (window.__amFordUnlockPopupOpen) return true;
    if (document.getElementById('custom-price-popup')) return true;
    if (document.getElementById('otp-vdp-popup')) return true;
    return false;
  }

  function removeTradeOfferOverlay() {
    var overlay = document.getElementById('custom-trade-offer-popup');
    if (overlay && overlay.parentNode) overlay.parentNode.removeChild(overlay);
    document.body.style.overflow = '';
  }

  function postTradeOfferContext(iframe, pageType, pageUrl) {
    if (!iframe || !iframe.contentWindow) return;
    try {
      iframe.contentWindow.postMessage({
        type: 'OTP_EMBED_CONTEXT',
        page_type: pageType,
        pageUrl: pageUrl
      }, '*');
    } catch (err) {}
  }

  function showTradeOfferPopup() {
    var pageType = getPageType();
    if (pageType !== 'Home' && pageType !== 'SRP') return;
    if (isUnlockPopupOpen()) return;

    removeTradeOfferOverlay();

    var pageUrl = window.location.href;
    var params =
      '?page_type=' + encodeURIComponent(pageType) +
      '&page_url=' + encodeURIComponent(pageUrl) +
      '&apiBase=' + encodeURIComponent(POPUP_ORIGIN);

    var overlay = document.createElement('div');
    overlay.id = 'custom-trade-offer-popup';
    overlay.style.cssText =
      'position:fixed;top:0;left:0;width:100%;height:100%;' +
      'background:linear-gradient(165deg,rgba(5,33,79,0.52) 0%,rgba(15,23,42,0.58) 100%);' +
      'backdrop-filter:saturate(115%) blur(12px);' +
      '-webkit-backdrop-filter:saturate(115%) blur(12px);' +
      'z-index:2147483647;display:flex;align-items:center;justify-content:center;';

    var box = document.createElement('div');
    box.style.cssText =
      'position:relative;width:100%;max-width:1080px;height:100vh;' +
      'background:transparent;overflow:hidden;';

    var loader = document.createElement('div');
    loader.style.cssText =
      'position:absolute;top:50%;left:50%;transform:translate(-50%,-50%);' +
      'width:38px;height:38px;border:3px solid rgba(255,255,255,0.25);' +
      'border-top-color:#ffffff;border-radius:50%;z-index:1;' +
      'animation:otpSpin 0.75s linear infinite;pointer-events:none;';

    var iframe = document.createElement('iframe');
    iframe.src = POPUP_ORIGIN + '/trade-offer' + params;
    iframe.setAttribute('allow', 'same-origin');
    iframe.style.cssText =
      'width:100%;height:100%;border:none;display:block;background:transparent;overflow:hidden;opacity:0;transition:opacity 0.2s ease;';

    iframe.addEventListener('load', function () {
      if (loader.parentNode) loader.parentNode.removeChild(loader);
      iframe.style.opacity = '1';
      postTradeOfferContext(iframe, pageType, pageUrl);
      setTimeout(function () { postTradeOfferContext(iframe, pageType, pageUrl); }, 250);
      setTimeout(function () { postTradeOfferContext(iframe, pageType, pageUrl); }, 800);
    });

    box.appendChild(loader);

    overlay.addEventListener('click', function (ev) {
      if (ev.target === overlay) removeTradeOfferOverlay();
    });

    var escHandler = function (ev) {
      if (ev.key === 'Escape') {
        removeTradeOfferOverlay();
        document.removeEventListener('keydown', escHandler);
      }
    };
    document.addEventListener('keydown', escHandler);

    window.addEventListener('message', function (e) {
      var originCleaned = e.origin.replace(/\/$/, '');
      var targetCleaned = POPUP_ORIGIN.replace(/\/$/, '');
      if (originCleaned !== targetCleaned) return;
      if ((e.data && e.data.type === 'TRADE_OFFER_CLOSE') || e.data === 'close-trade-offer' ||
          (e.data && e.data.type === 'TRADE_OFFER_SUBMITTED')) {
        removeTradeOfferOverlay();
        document.removeEventListener('keydown', escHandler);
      }
    });

    box.appendChild(iframe);
    overlay.appendChild(box);
    document.body.appendChild(overlay);
    document.body.style.overflow = 'hidden';
  }

  function initTimedTradeOffer() {
    clearOldEnquiry();
    clearEnquiryCookies();
    var pageType = getPageType();
    if (pageType !== 'Home' && pageType !== 'SRP') return;
    var delaySeconds = 18;
    setTimeout(function () {
      if (isUnlockPopupOpen()) return;
      showTradeOfferPopup();
    }, delaySeconds * 1000);
  }

  window.TradeOfferPopup = { open: showTradeOfferPopup, close: removeTradeOfferOverlay };

  if (document.readyState === 'complete') {
    initTimedTradeOffer();
  } else {
    window.addEventListener('load', initTimedTradeOffer);
  }
})();
</script>
